        /**
         * C++ is_alphabet logic: Checks if both characters are alphabetic
         */
        function is_alphabet(a, b) {
            if (!a || !b) return false;
            
            const isLower = c => c >= 'a' && c <= 'z';
            const isUpper = c => c >= 'A' && c <= 'Z';

            return (
                (isLower(a) && isLower(b)) ||
                (isUpper(a) && isUpper(b)) ||
                (isLower(a) && isUpper(b)) ||
                (isUpper(a) && isLower(b))
            );
        }

        /**
         * C++ logic: tolower(char) - 96
         */
        function getAlphabetPosition(char) {
            if (!char) return 0;
            return char.toLowerCase().charCodeAt(0) - 96;
        }
